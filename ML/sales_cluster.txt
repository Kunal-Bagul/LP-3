import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Step 1: Load dataset
df = pd.read_csv('sales_data_sample.csv', encoding='latin1')

# Step 2: Check dataset
print(df.head())
print(df.isnull().sum())

# Step 3: Select features
X = df[['QUANTITYORDERED', 'PRICEEACH', 'SALES']]

# Step 4: Standardize features
scaler = StandardScaler()
X_s = scaler.fit_transform(X)

# Step 5: Find optimal clusters using Elbow Method
wcss = []

for k in range(1, 11):
    kmeans = KMeans(
        n_clusters=k,
        random_state=42,
        n_init=10
    )
    kmeans.fit(X_s)
    wcss.append(kmeans.inertia_)

# Step 6: Plot Elbow graph
plt.plot(range(1, 11), wcss, marker='o')
plt.title('Elbow Method')
plt.xlabel('Number of Clusters')
plt.ylabel('WCSS')
plt.grid(True)
plt.show()

# Step 7: Apply K-Means
# Select 3 only if supported by your elbow graph
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
df['cluster'] = kmeans.fit_predict(X_s)

# Step 8: Display results
print(df[['QUANTITYORDERED', 'PRICEEACH',
          'SALES', 'cluster']].head())

# Step 9: Visualize clusters
plt.scatter(
    df['QUANTITYORDERED'],
    df['SALES'],
    c=df['cluster'],
    cmap='viridis'
)

plt.xlabel('QUANTITYORDERED')
plt.ylabel('SALES')
plt.title('K-Means Clustering')
plt.colorbar(label='Cluster')
plt.grid(True)
plt.show()
